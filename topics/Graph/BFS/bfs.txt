vector<vector<int>> nodes;
vector<bool> visited;
int n,m;
vector<int> levels;
void bfs(int start) {
    queue<int> q;
    visited[start] = 1;
    levels[start] = 1;
    q.push(start);
    while(! q.empty()) {
        int cur = q.front();
        q.pop();
        for(auto ch : nodes[cur]) {
            if(visited[ch] == 0) {
                q.push(ch);
                visited[ch] = 1;
                levels[ch] = levels[cur] + 1;
            }
        }
    }
}
int main() {
    ana_Yano;
    cin>>n;
    nodes.assign(n + 1,vector<int>());
    visited.assign(n + 1,0);
    levels.assign(n + 1,0);
    
    for(int i = 0,u,v;i<n-1;i++) {
        cin>>u>>v;
        nodes[u].push_back(v);
        nodes[v].push_back(u);
    }
    bfs(1);
    return 0;
}
